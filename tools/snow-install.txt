            {
                foreach (string line in File.ReadAllLines(path))
                {
                    string[] kv = line.Split('=');
                    if (kv.Length != 2) continue;

                        case "tips": Tips = v == 1; break;

                        case "fps": Fps = v == 1; break;

        public void SetLevel(int l)
        {
            S.Level = l;
            int n = l == 0 ? 110 : l == 1 ? 240 : 520;
            while (flakes.Count < n) flakes.Add(NewFlake(true));
            if (flakes.Count > n) flakes.RemoveRange(n, flakes.Count - n);
        }

        Flake NewFlake(bool anywhere)
        {
            Flake f = new Flake();
            f.Depth = rnd.NextDouble();
            f.Crystal = rnd.NextDouble() < 0.16;
            f.R = (f.Crystal ? 4.5 : 1.2) + f.Depth * (f.Crystal ? 5.5 : 3.2);
            f.X = rnd.NextDouble() * (Width + 200) - 100;
            f.Y = anywhere ? rnd.NextDouble() * Height : -10 - rnd.NextDouble() * 60;
            f.Vy = 22 + f.Depth * 55 + rnd.NextDouble() * 18;
            f.Phase = rnd.NextDouble() * 6.28;
            f.Amp = 8 + rnd.NextDouble() * 22;
            f.Rot = rnd.NextDouble() * 360;
            f.RotSpeed = (rnd.NextDouble() - 0.5) * 90;
            return f;
        }

        double GroundAt(double x) { int i = (int)(x / COL); return i < 0 || i >= ground.Length ? 0 : ground[i]; }

        void AddSnow(double x, double amount)
        {
            int i = (int)(x / COL);
            for (int k = -2; k <= 2; k++)
            {
                int j = i + k;
                if (j < 0 || j >= ground.Length) continue;
                ground[j] = Math.Min(MAXG, ground[j] + amount * (k == 0 ? 0.5 : k == -1 || k == 1 ? 0.2 : 0.05));
            }
        }

        double CapAt(Cap c, double x)
        {
            int i = (int)((x - c.L) / COL);
            if (i < 0 || i >= c.H.Length) return -1;
            return c.H[i];
        }

        // высота поверхности (верх снега) под точкой x для опоры: окно или пол
        double SurfaceY(IntPtr anchor, double x)
        {
            Cap c;
            if (anchor != IntPtr.Zero && caps.TryGetValue(anchor, out c)) return c.T - Math.Max(0, CapAt(c, x));
            return groundY - GroundAt(x);
        }

        // ---------- шаг симуляции ----------
        public void Step()
        {
            DateTime now = DateTime.Now;
            double dt = Math.Min(0.1, (now - last).TotalSeconds);
            last = now;
            if (S.Paused) return;
            time += dt; Clock.Now = time;
            tick++;
            if (tick % 30 == 1) Housekeeping();
            if (tick % 10 == 0) Front.EnsureClickThrough();
            if (hidden) return;

            // ветер
            if (time > nextGust)
            {
                double baseW = S.Wind == 2 ? -70 : S.Wind == 3 ? 70 : 0;
                double gust = S.Wind == 1 ? 0 : (rnd.NextDouble() - 0.4) * (S.Level == 2 ? 140 : 60);
                if (S.Wind >= 2) gust = (rnd.NextDouble() - 0.5) * 40;
                windTarget = baseW + gust;
                nextGust = time + 3 + rnd.NextDouble() * 6;
            }
            wind += (windTarget - wind) * Math.Min(1, dt * 0.6);

            Native.POINT p;
            pmx = mx; pmy = my;
            if (Native.GetCursorPos(out p)) { mx = p.X / Scale; my = p.Y / Scale; }
            double mdx = mx - pmx, mdy = my - pmy;
            if (Math.Abs(mdx) > 400 || Math.Abs(mdy) > 400) { mdx = mdy = 0; }
            mSpeed = Math.Sqrt(mdx * mdx + mdy * mdy) / Math.Max(dt, 0.001);

            StepFlakes(dt);
            Settle();
            if (S.Tool == 1) Mitten(dt, mdx, 4, 0.35, 1);
            if (S.Tool == 3) Mitten(dt, mdx, 9, 0.9, 2.2);
            if (S.Tool == 2) Snowball(dt, mdx);
            StepChunks(dt);
            StepTractor(dt);
            StepMen(dt);
            StepSanta(dt);

            frames++;
            if ((now - fpsAt).TotalSeconds >= 1) { fps = frames; frames = 0; fpsAt = now; }
            Front.V.InvalidateVisual();
        }

        void StepFlakes(double dt)
        {
            foreach (Flake f in flakes)
            {
                f.Phase += dt * (0.8 + f.Depth);
                double vx = wind * (0.4 + f.Depth * 0.8) + Math.Cos(f.Phase) * f.Amp * 0.6;
                if (S.Mouse)
                {
                    double dx = f.X - mx, dy = f.Y - my, d2 = dx * dx + dy * dy;
                    if (d2 < 120 * 120 && d2 > 1)
                    {
                        double d = Math.Sqrt(d2), push = (1 - d / 120) * 420;
                        f.Vx += dx / d * push * dt;
                        f.Y += dy / d * push * dt * 0.5;
                    }
                }
                f.Vx *= Math.Pow(0.15, dt);
                f.X += (vx + f.Vx) * dt;
                f.Y += f.Vy * dt * (S.Level == 2 ? 1.35 : 1);
                f.Rot += f.RotSpeed * dt;
                if (f.X < -120) f.X += Width + 220;
                if (f.X > Width + 120) f.X -= Width + 220;

                bool landed = false;
                if (f.Depth > 0.45)
                {
                    if (S.Windows)
                    {
                        foreach (Cap c in caps.Values)
                        {
                            if (f.X < c.L || f.X > c.R) continue;
                            int ci = (int)((f.X - c.L) / COL);
                            if (ci < 0 || ci >= c.H.Length || c.H[ci] < 0) continue;
                            double top = c.T - c.H[ci];
                            if (f.Y >= top && f.Y < top + 14)
                            {
                                for (int k = -1; k <= 1; k++)
                                {
                                    int j = ci + k;
                                    if (j >= 0 && j < c.H.Length && c.H[j] >= 0)
                                        c.H[j] = Math.Min(18, c.H[j] + f.R * (k == 0 ? 1.0 : 0.4));
                                }
                                landed = true; break;
                            }
                        }
                    }
                    if (!landed && f.Y >= groundY - GroundAt(f.X)) { AddSnow(f.X, f.R * 2.0); landed = true; }
                }
                if (landed || f.Y > Height + 20)
                {
                    Flake n = NewFlake(false);
                    f.X = n.X; f.Y = n.Y; f.Vy = n.Vy; f.Vx = 0; f.R = n.R; f.Depth = n.Depth; f.Crystal = n.Crystal;
                }
            }
        }

        void Settle()
        {
            for (int i = 1; i < ground.Length - 1; i++)
            {
                double avg = (ground[i - 1] + ground[i + 1]) * 0.5;
                if (Math.Abs(ground[i] - avg) > 1.2) ground[i] += (avg - ground[i]) * 0.25;
            }
            foreach (Cap c in caps.Values)
                for (int i = 1; i < c.H.Length - 1; i++)
                {
                    if (c.H[i] < 0 || c.H[i - 1] < 0 || c.H[i + 1] < 0) continue;
                    c.H[i] += ((c.H[i - 1] + c.H[i + 1]) * 0.5 - c.H[i]) * 0.2;
                }
        }

        // варежка: проводишь курсором по сугробу — снег разлетается
        // варежка и лопата: проводишь курсором по снегу — он разлетается
        void Mitten(double dt, double mdx, int reach, double power, double chunk)
        {
            if (mSpeed < 40) return;
            double amount = Math.Min(4 + power * 6, mSpeed * dt * power);
            double top = groundY - GroundAt(mx);
            if (my > top - 18 && my < groundY + 6) SweepCols(ground, 0, mx, amount, mdx, groundY, reach, chunk);
            foreach (Cap c in caps.Values)
            {
                if (mx < c.L - 10 || mx > c.R + 10) continue;
                double h = Math.Max(0, CapAt(c, mx));
                if (my > c.T - h - 18 && my < c.T + 8) SweepCols(c.H, c.L, mx, amount, mdx, c.T, reach, chunk);
            }
        }

        void SweepCols(double[] h, double left, double x, double amount, double mdx, double baseY, int reach, double chunk)
        {
            int i0 = (int)((x - left) / COL);
            for (int k = -reach; k <= reach; k++)
            {
                int j = i0 + k;
                if (j < 0 || j >= h.Length || h[j] <= 0.3) continue;
                double take = Math.Min(h[j], amount * (1 - Math.Abs(k) / (reach + 1.0)));
                h[j] -= take;
                if (take > 0.6 && rnd.NextDouble() < 0.5 && chunks.Count < 800)
                    chunks.Add(new Chunk { X = left + j * COL, Y = baseY - h[j] - 2, Vx = mdx * 8 + (rnd.NextDouble() - 0.5) * 80, Vy = -120 - rnd.NextDouble() * 140 * chunk, R = (1.4 + rnd.NextDouble() * 2) * chunk, Life = 1.6 });
            }
        }

        // насколько курсор близко к снегу (для картинки инструмента у курсора)
        double toolFade;
        bool CursorNearSnow()
        {
            if (my > groundY - GroundAt(mx) - 90 && my < groundY + 10) return true;
            foreach (Cap c in caps.Values)
                if (mx > c.L - 20 && mx < c.R + 20 && my > c.T - 90 && my < c.T + 12) return true;
            return false;
        }

        // снежок как во Flake: зажми левую кнопку на сугробе и веди — шар катится и растёт;
        // отпусти — шар встанет. Шар можно взять и перенести на другой — будет снеговик.
        bool wasDown; bool carrying;
        void Snowball(double dt, double mdx)
        {
            bool down = (Native.GetAsyncKeyState(0x01) & 0x8000) != 0;
            if (down && !wasDown) StartBall();
            wasDown = down;
            if (rolling == null) return;
            if (!down) { PlaceBall(); return; }
            double r = rolling.R[0];
            IntPtr anchor; double sy, h;
            bool onSnow = NearSurface(mx, my + r * 0.5, out anchor, out sy, out h) && my > sy - r * 2.2 - 30;
            double dx = mx - rolling.X;
            rolling.X = mx;
            if (onSnow)
            {
                carrying = false;
                rolling.Anchor = anchor;
                rollRot += dx / Math.Max(4, r) * 57.3;
                if (Math.Abs(dx) > 0.3)
                {
                    double got = Dig(anchor, mx, r, Math.Abs(dx) * 0.12);
                    rolling.R[0] = Math.Min(44, r + got * 0.09 + Math.Abs(dx) * 0.012);
                }
            }
            else carrying = true;   // шар подняли — несём его в руке
        }

        void StartBall()
        {
            // взять уже стоящий шар (верхний в стопке)
            for (int i = men.Count - 1; i >= 0; i--)
            {
                Man m = men[i];
                if (m.Melt > 0 || !double.IsNaN(m.FallY)) continue;
                double topR = m.R[m.R.Count - 1], cy = TopY(m) + topR;
                if ((mx - m.X) * (mx - m.X) + (my - cy) * (my - cy) < topR * topR * 1.3)
                {
                    rolling = new Man { X = m.X, Anchor = m.Anchor };
                    rolling.R.Add(topR);
                    m.R.RemoveAt(m.R.Count - 1);
                    if (m.R.Count == 0) men.RemoveAt(i); else m.DecorateAt = double.MaxValue;
                    carrying = true;
                    return;
                }
            }
            IntPtr anchor; double sy, h;
            if (!NearSurface(mx, my, out anchor, out sy, out h)) return;
            rolling = new Man { X = mx, Anchor = anchor };
            rolling.R.Add(6);
            carrying = false;
        }

        double Dig(IntPtr anchor, double x, double r, double amount)
        {
            double[] h = ground; double left = 0;
            Cap c;
            if (anchor != IntPtr.Zero && caps.TryGetValue(anchor, out c)) { h = c.H; left = c.L; }
            int i0 = (int)((x - left) / COL), w = Math.Max(2, (int)(r / COL));
            double got = 0;
            for (int k = -w; k <= w; k++)
            {
                int j = i0 + k;
                if (j < 0 || j >= h.Length || h[j] <= 0.5) continue;
                double take = Math.Min(h[j] - 0.5, amount);
                h[j] -= take; got += take;
            }
            return got;
        }

        bool NearSurface(double x, double y, out IntPtr anchor, out double sy, out double h)
        {
            foreach (KeyValuePair<IntPtr, Cap> kv in caps)
            {
                Cap c = kv.Value;
                if (x < c.L || x > c.R) continue;
                double ch = CapAt(c, x);
                if (ch < 0) continue;
                if (y > c.T - ch - 40 && y < c.T + 6) { anchor = kv.Key; sy = c.T - ch; h = ch; return true; }
            }
            double g = GroundAt(x);
            anchor = IntPtr.Zero; sy = groundY - g; h = g;
            return y > sy - 40 && y < groundY + 6;
        }

        void PlaceBall()
        {
            Man b = rolling; rolling = null; carrying = false;
            double r = b.R[0];
            if (r < 7) { Puff(b.X, my, 6); return; }
            Man best = null;
            foreach (Man m in men)
            {
                if (m.Melt > 0 || !double.IsNaN(m.FallY) || m.R.Count >= 3) continue;
                if (Math.Abs(m.X - b.X) < m.R[m.R.Count - 1] + r * 0.6 && Math.Abs(my - TopY(m)) < 140) best = m;
            }
            if (best != null)
            {
                best.R.Add(Math.Min(r, best.R[best.R.Count - 1] * 1.05));
                best.DecorateAt = best.R.Count >= 3 ? time + 0.15 : time + 1.6;
                best.WaveAt = time + 1;
                Sparkle(best.X, TopY(best), 16);
                return;
            }
            b.Hat = rnd.Next(HatColors.Length + 1); b.Scarf = rnd.Next(ScarfColors.Length);
            // роняем на поверхность под курсором
            b.Anchor = IntPtr.Zero;
            double surf = SurfaceY(IntPtr.Zero, b.X);
            if (my < surf - r * 2) { b.FallY = my + r; b.FallV = 0; }
            foreach (KeyValuePair<IntPtr, Cap> kv in caps)
            {
                Cap c = kv.Value;
                if (b.X > c.L && b.X < c.R && CapAt(c, b.X) >= 0 && Math.Abs(my + r - (c.T - CapAt(c, b.X))) < r * 1.5) { b.Anchor = kv.Key; b.FallY = double.NaN; break; }
            }
            men.Add(b);
            if (men.Count > 14) men.RemoveAt(0);
        }

        void Puff(double x, double y, int n)
        {
            for (int k = 0; k < n; k++)
                chunks.Add(new Chunk { X = x, Y = y, Vx = (rnd.NextDouble() - 0.5) * 160, Vy = -80 - rnd.NextDouble() * 120, R = 1.5 + rnd.NextDouble() * 2, Life = 1.2 });
        }

        void Sparkle(double x, double y, int n)
        {
            for (int k = 0; k < n; k++)
            {
                double a = rnd.NextDouble() * 6.28, v = 40 + rnd.NextDouble() * 90;
                sparks.Add(new Spark { X = x, Y = y, Vx = Math.Cos(a) * v, Vy = Math.Sin(a) * v - 30, Life = 0.9 + rnd.NextDouble() * 0.5, Max = 1.4 });
            }
        }

        void StepChunks(double dt)
        {
            for (int i = chunks.Count - 1; i >= 0; i--)
            {
                Chunk c = chunks[i];
                c.Vy += 600 * dt; c.X += c.Vx * dt; c.Y += c.Vy * dt; c.Life -= dt;
                if (c.Vy > 0 && c.Y >= groundY - GroundAt(c.X)) { AddSnow(c.X, c.R * 0.3); chunks.RemoveAt(i); continue; }
                if (c.Life <= 0 || c.Y > Height) chunks.RemoveAt(i);
            }
            for (int i = sparks.Count - 1; i >= 0; i--)
            {
                Spark s = sparks[i];
                s.Vy += 30 * dt; s.X += s.Vx * dt; s.Y += s.Vy * dt; s.Vx *= Math.Pow(0.3, dt); s.Life -= dt;
                if (s.Life <= 0) sparks.RemoveAt(i);
            }
        }

        void StepTractor(double dt)
        {
            if (double.IsNaN(tractorX)) return;
            tractorX += 190 * dt;
            double plow = tractorX + 162;
            int i0 = (int)(plow / COL);
            for (int k = 0; k < 6; k++)
            {
                int j = i0 + k;
                if (j < 0 || j >= ground.Length) continue;
                double take = Math.Max(0, ground[j] - 0.5) * 0.92;
                plowPile = Math.Min(400, plowPile + take);
                ground[j] -= take;
            }
            // снег перед отвалом вылетает струёй вперёд-вверх
            int nSpray = (int)Math.Min(4, plowPile / 25);
            for (int k = 0; k < nSpray && chunks.Count < 800; k++)
            {
                chunks.Add(new Chunk { X = plow + 6, Y = groundY - 14 - rnd.NextDouble() * 10, Vx = 160 + rnd.NextDouble() * 220, Vy = -260 - rnd.NextDouble() * 220, R = 1.8 + rnd.NextDouble() * 2.8, Life = 1.6 });
                plowPile -= 6;
            }
            plowPile *= Math.Pow(0.6, dt);
            foreach (Man m in men) if (m.Anchor == IntPtr.Zero && double.IsNaN(m.FallY) && Math.Abs(m.X - plow) < 30 && m.Melt == 0) m.Melt = 0.001;
            if (tractorX > Width + 160) tractorX = double.NaN;
        }

        void StepMen(double dt)
        {
            for (int i = men.Count - 1; i >= 0; i--)
            {
                Man m = men[i];
                if (!double.IsNaN(m.FallY))   // окно ушло — снеговик падает вниз
                {
                    m.FallV += 900 * dt; double before = m.FallY; m.FallY += m.FallV * dt;
                    bool landed = false;
                    foreach (KeyValuePair<IntPtr, Cap> kv in caps)
                    {
                        Cap c = kv.Value; double ch = CapAt(c, m.X);
                        if (m.X > c.L && m.X < c.R && ch >= 0 && before <= c.T - ch && m.FallY >= c.T - ch) { m.Anchor = kv.Key; m.FallY = double.NaN; m.FallV = 0; landed = true; break; }
                    }
                    if (landed) continue;
                    double g = groundY - GroundAt(m.X);
                    if (m.FallY >= g) { m.FallY = double.NaN; m.FallV = 0; m.Anchor = IntPtr.Zero; Puff(m.X, g, 8); }
                    continue;
                }
                if (m.Anchor != IntPtr.Zero && !caps.ContainsKey(m.Anchor)) { m.FallY = BaseY(m); m.FallV = 0; continue; }
                if (m.Melt > 0)
                {
                    m.Melt += dt * 0.8;
                    if (m.Melt > 1) { Puff(m.X, BaseY(m) - 30, 14); men.RemoveAt(i); }
                }
                else if (m.Decorated && time > m.WaveAt + 7 + rnd.NextDouble() * 10) m.WaveAt = time;
            }
        }

        void StepSanta(double dt)
        {
            if (S.Santa && double.IsNaN(santaX) && time > nextSanta) CallSanta();
            if (double.IsNaN(santaX)) return;
            santaX -= 240 * dt;
            if (rnd.NextDouble() < 0.9)
                sparks.Add(new Spark { X = santaX + 0.85 * 440, Y = santaY + 0.85 * 100 + (rnd.NextDouble() - 0.5) * 14, Vx = 40 + rnd.NextDouble() * 40, Vy = (rnd.NextDouble() - 0.3) * 30, Life = 1.2 + rnd.NextDouble(), Max = 2.2 });
            if (santaX < -520) { santaX = double.NaN; nextSanta = time + 420 + rnd.NextDouble() * 600; }
        }

            if (fg != IntPtr.Zero && fg != Front.Hwnd && fg != Native.GetShellWindow() && Native.GetWindowRect(fg, out fr))

            }

            {
                hidden = full;

            {

        }

        int blockedCount;
        void CheckClickThrough()
        {
            Native.POINT p;
            if (!Native.GetCursorPos(out p)) return;
            IntPtr h = Native.WindowFromPoint(p);
            if (h != IntPtr.Zero && Native.GetAncestor(h, 2) == Front.Hwnd) blockedCount++; else blockedCount = 0;
            if (blockedCount >= 2)
            {
                blockedCount = 0;
                if (!S.Compat) { S.Compat = true; S.Save(); Program.Restart(); }
                else { S.Paused = true; Front.Visibility = Visibility.Hidden; if (Tip != null) Tip("Снег мешал нажимать на окна — я его приостановил. Включи снова в панели (Пауза)."); }
            }
        }

        static string ClassOf(IntPtr h) { StringBuilder sb = new StringBuilder(64); Native.GetClassName(h, sb, 64); return sb.ToString(); }

        void ScanWindows()
        {
            foreach (Cap c in caps.Values) c.Seen = false;
            List<Native.RECT> above = new List<Native.RECT>();
            Geometry mask = new RectangleGeometry(new Rect(0, 0, Width, Height));
            Native.EnumWindows(delegate (IntPtr h, IntPtr p)
            {
                if (h == Front.Hwnd || !Native.IsWindowVisible(h) || Native.IsIconic(h)) return true;
                int cloaked;
                if (Native.DwmGetWindowAttribute(h, 14, out cloaked, 4) == 0 && cloaked != 0) return true;
                string cls = ClassOf(h);
                if (cls == "Progman" || cls == "WorkerW" || cls == "Shell_TrayWnd" || cls == "Shell_SecondaryTrayWnd") return true;
                Native.RECT r;
                if (Native.DwmGetWindowAttribute(h, 9, out r, 16) != 0) Native.GetWindowRect(h, out r);
                if ((Native.GetWindowLong(h, -20) & 0x20) != 0) return true;   // прозрачные оверлеи других программ не считаем
                if (r.R > r.L && r.B > r.T && Native.GetWindowTextLength(h) > 0)
                    mask = Geometry.Combine(mask, new RectangleGeometry(new Rect(r.L / Scale, r.T / Scale, (r.R - r.L) / Scale, (r.B - r.T) / Scale)), GeometryCombineMode.Exclude, null);
                if (!S.Windows) { above.Add(r); return true; }
                if (r.R - r.L < 160 * Scale || r.B - r.T < 80 * Scale || Native.GetWindowTextLength(h) == 0) { above.Add(r); return true; }
                double L = r.L / Scale, T = r.T / Scale, R = r.R / Scale;
                if (T > 30 && T < groundY - 40 && L < Width && R > 0)
                {
                    Cap c;
                    if (!caps.TryGetValue(h, out c) || Math.Abs(c.L - L) > 1 || Math.Abs(c.T - T) > 1 || Math.Abs(c.R - R) > 1)
                    {
                        if (c != null) { Crumble(c); DropMen(h); }
                        c = new Cap { L = L, T = T, R = R, H = new double[(int)((R - L) / COL) + 1] };
                        caps[h] = c;
                    }
                    c.Seen = true;
                    for (int i = 0; i < c.H.Length; i++)
                    {
                        int px = (int)((c.L + i * COL) * Scale), py = r.T - 2;
                        bool covered = false;
                        foreach (Native.RECT a in above) if (px >= a.L && px < a.R && py >= a.T && py < a.B) { covered = true; break; }
                        if (covered) { if (c.H[i] > 0) Spill(c.L + i * COL, c.T, c.H[i]); c.H[i] = -1; }
                        else if (c.H[i] < 0) c.H[i] = 0;
                    }
                }
                above.Add(r);
                return true;
            }, IntPtr.Zero);
            mask.Freeze();
            clip = mask;
            List<IntPtr> gone = new List<IntPtr>();
            foreach (KeyValuePair<IntPtr, Cap> kv in caps) if (!kv.Value.Seen) gone.Add(kv.Key);
            foreach (IntPtr h in gone) { Crumble(caps[h]); DropMen(h); caps.Remove(h); }
        }

        void DropMen(IntPtr h)
        {
            foreach (Man m in men) if (m.Anchor == h && double.IsNaN(m.FallY)) { m.FallY = BaseY(m); m.FallV = 0; }
            if (rolling != null && rolling.Anchor == h) rolling = null;
        }

        void Spill(double x, double y, double h)
        {
            if (h < 2 || chunks.Count > 700) return;

        }

        void DropAllCaps()
        {
            foreach (KeyValuePair<IntPtr, Cap> kv in caps) { Crumble(kv.Value); DropMen(kv.Key); }
            caps.Clear();
        }

        }

        public void AutoSnowman()
        {
            double best = -1, bestX = Width / 2;
            for (int t = 0; t < 50; t++)
            {
                double x = 90 + rnd.NextDouble() * (Width - 180);
                bool near = false;
                foreach (Man m in men) if (m.Anchor == IntPtr.Zero && Math.Abs(m.X - x) < 110) near = true;
                if (near) continue;
                double sum = 0;
                for (double k = x - 120; k <= x + 120; k += COL) sum += GroundAt(k);
                if (sum > best) { best = sum; bestX = x; }
            }
            for (double k = bestX - 100; k <= bestX + 100; k += COL) { int j = (int)(k / COL); if (j >= 0 && j < ground.Length) ground[j] *= 0.6; }
            double s = 0.85 + rnd.NextDouble() * 0.35;
            Man m2 = new Man { X = bestX, Born = time, Hat = rnd.Next(HatColors.Length + 1), Scarf = rnd.Next(ScarfColors.Length) };
            m2.R.Add(30 * s); m2.R.Add(22 * s); m2.R.Add(15 * s);
            m2.DecorateAt = time + 1.9; m2.WaveAt = time + 2.6;
            men.Add(m2);
            if (men.Count > 12) men.RemoveAt(0);
        }

        public void DrawFront(DrawingContext dc)
        {
            if (hidden) return;

            if (masked) foreach (Flake f in flakes) DrawFlake(dc, f);

            foreach (Chunk c in chunks) dc.DrawEllipse(glow[5], null, new Point(c.X, c.Y), c.R * 1.6, c.R * 1.6);

            {

            if (!masked) foreach (Flake f in flakes) if (f.Depth >= 0.45) DrawFlake(dc, f);
            if (S.Fps)
            {
                FormattedText ft = new FormattedText("FPS " + fps + "  окон " + caps.Count, System.Globalization.CultureInfo.CurrentCulture, FlowDirection.LeftToRight, new Typeface("Segoe UI"), 16, Brushes.White, Scale);
                dc.DrawRoundedRectangle(new SolidColorBrush(Color.FromArgb(150, 0, 0, 0)), null, new Rect(8, 40, ft.Width + 16, ft.Height + 8), 6, 6);
                dc.DrawText(ft, new Point(16, 44));
            }
            // варежка / лопата / снежок у курсора, когда он рядом со снегом
            bool near = S.Tool != 0 && rolling == null && (CursorNearSnow() || time < ToolShowUntil);
            toolFade = Math.Max(0, Math.Min(1, toolFade + (near ? 0.15 : -0.1)));
            if (toolFade > 0.01)
            {
                double tilt = Math.Max(-25, Math.Min(25, (mx - pmx) * 1.5));
                dc.PushOpacity(toolFade);
                if (S.Tool == 1) art.DrawMitten(dc, mx, my, tilt);
                else if (S.Tool == 3) art.DrawShovel(dc, mx, my, tilt);

        void DrawFlake(DrawingContext dc, Flake f)
        {
            if (f.Crystal)
            {
                Matrix m = Matrix.Identity; m.Scale(f.R, f.R); m.Rotate(f.Rot); m.Translate(f.X, f.Y);
                dc.PushTransform(new MatrixTransform(m));

                dc.Pop(); dc.Pop();

        void DrawGround(DrawingContext dc)
        {
            StreamGeometry g = new StreamGeometry();
            using (StreamGeometryContext c = g.Open())
            {
                c.BeginFigure(new Point(0, groundY + 2), true, true);
                for (int i = 0; i < ground.Length; i++) c.LineTo(new Point(i * COL, groundY - ground[i] + 1), true, true);
                c.LineTo(new Point(Width, groundY + 2), true, false);
            }
            g.Freeze();
            dc.DrawGeometry(snowFill, snowEdge, g);
            for (int i = 3; i < ground.Length; i += 9)
            {
                if (ground[i] < 4) continue;
                if (Math.Sin(time * 3 + i * 1.7) > 0.85) dc.DrawEllipse(glow[5], null, new Point(i * COL, groundY - ground[i] + 4), 2.2, 2.2);
            }
        }

        void DrawCap(DrawingContext dc, Cap c)
        {
            int i = 0;
            while (i < c.H.Length)
            {
                while (i < c.H.Length && c.H[i] <= 0.3) i++;
                if (i >= c.H.Length) break;
                int s = i;
                StreamGeometry g = new StreamGeometry();
                using (StreamGeometryContext k = g.Open())
                {
                    k.BeginFigure(new Point(c.L + s * COL, c.T + 1), true, true);
                    while (i < c.H.Length && c.H[i] > 0.3) { k.LineTo(new Point(c.L + i * COL, c.T - c.H[i]), true, true); i++; }
                    k.LineTo(new Point(c.L + (i - 1) * COL, c.T + 1), true, false);
                }
                g.Freeze();
                dc.DrawGeometry(capFill, snowEdge, g);
            }
        }

        double BaseY(Man m)
        {
            if (!double.IsNaN(m.FallY)) return m.FallY;
            return SurfaceY(m.Anchor, m.X) + Math.Min(6, m.R[0] * 0.2);
        }

        double TopY(Man m)  // верх верхнего шара — та же раскладка, что в DrawMan
        {
            double y = BaseY(m), c = y;

        void DrawBall(DrawingContext dc, double x, double y, double r, double rot)
        {

        void DrawMan(DrawingContext dc, Man m)
        {
            ManBrushes();
            double age = time - m.Born;
            double baseY = BaseY(m);
            double shrink = 1 - Math.Min(1, m.Melt) * 0.9;
            int n = m.R.Count;
            double[] ys = new double[n], rs = new double[n];
            double y = baseY;
            for (int i = 0; i < n; i++)
            {
                rs[i] = m.R[i] * shrink;
                y -= rs[i] * (i == 0 ? 0.95 : 0.82);
                ys[i] = y;
                y -= rs[i] * 0.82;
            }
            dc.PushOpacity(1 - Math.Min(1, m.Melt));
            dc.DrawEllipse(ballShadow, null, new Point(m.X, ys[0] + rs[0] * 0.92), rs[0] * 1.05, rs[0] * 0.22);
            bool decor = m.Decorated;
            double pop = decor ? Ease(Clamp((time - m.DecorateAt) / 0.5)) : 0;
            if (decor) { dc.PushOpacity(pop); DrawArms(dc, m, ys, rs); dc.Pop(); }
            for (int i = 0; i < n; i++)
            {
                double pr = m.Born < 0 ? 1 : Clamp((age - (i == 0 ? 0 : i == 1 ? 0.8 : 1.4)) / (i == 0 ? 0.8 : 0.6));
                if (pr <= 0) continue;
                double ox = i == 0 ? -(1 - Ease(pr)) * 140 : 0, oy = i == 0 ? 0 : -(1 - Ease(pr)) * 70;
                DrawBall(dc, m.X + ox, ys[i] + oy, rs[i] * (i == 0 ? Ease(pr) : Math.Max(0.3, Ease(pr))), 0);
            }
            if (decor) { dc.PushOpacity(pop); DrawDecor(dc, m, ys, rs); dc.Pop(); }
            dc.Pop();
        }

        void Branch(DrawingContext dc, Pen pen, double sx, double sy, double deg, double len, int flip, double s)
        {
            double a = deg * Math.PI / 180;
            double ex = sx + Math.Cos(a) * len, ey = sy + Math.Sin(a) * len;
            StreamGeometry g = new StreamGeometry();
            using (StreamGeometryContext c = g.Open())
            {
                c.BeginFigure(new Point(sx, sy), false, false);
                c.QuadraticBezierTo(new Point((sx + ex) / 2, (sy + ey) / 2 - 3 * s), new Point(ex, ey), true, true);
                double mx1 = sx + Math.Cos(a) * len * 0.62, my1 = sy + Math.Sin(a) * len * 0.62, b1 = a + flip * 0.75;
                c.BeginFigure(new Point(mx1, my1), false, false);
                c.LineTo(new Point(mx1 + Math.Cos(b1) * len * 0.28, my1 + Math.Sin(b1) * len * 0.28), true, true);
                double mx2 = sx + Math.Cos(a) * len * 0.85, my2 = sy + Math.Sin(a) * len * 0.85, b2 = a - flip * 0.6;
                c.BeginFigure(new Point(mx2, my2), false, false);
                c.LineTo(new Point(mx2 + Math.Cos(b2) * len * 0.18, my2 + Math.Sin(b2) * len * 0.18), true, true);
            }
            dc.DrawGeometry(null, pen, g);
        }

        void DrawArms(DrawingContext dc, Man m, double[] ys, double[] rs)
        {
            int n = rs.Length;
            double br = rs[n - 2], by = ys[n - 2], s = rs[n - 1] / 16.0;
            Pen pen = new Pen(stick, 2.6 * s); pen.StartLineCap = pen.EndLineCap = PenLineCap.Round; pen.LineJoin = PenLineJoin.Round;
            double ay = by - br * 0.25;
            double wave = time > m.WaveAt && time - m.WaveAt < 2 ? Math.Sin((time - m.WaveAt) * 12) * 28 : 0;
            Branch(dc, pen, m.X - br * 0.85, ay, 205, br * 1.25, -1, s);
            Branch(dc, pen, m.X + br * 0.85, ay, -35 + wave, br * 1.25, 1, s);
        }

        void DrawGarland(DrawingContext dc)
        {
            if (bulbHalo == null)
            {
                bulbHalo = new Brush[BulbColors.Length]; bulbCore = new Brush[BulbColors.Length];
                for (int i = 0; i < BulbColors.Length; i++)
                {
                    Color c = BulbColors[i];
                    bulbHalo[i] = new RadialGradientBrush(Color.FromArgb(170, c.R, c.G, c.B), Color.FromArgb(0, c.R, c.G, c.B)); bulbHalo[i].Freeze();
                    bulbCore[i] = new SolidColorBrush(c); bulbCore[i].Freeze();
                }
                wirePen = new Pen(new SolidColorBrush(Color.FromArgb(200, 30, 60, 30)), 1.6); wirePen.Freeze();

            int n = (int)(Width / 70) + 1;
            StreamGeometry g = new StreamGeometry();

                for (int i = 0; i < n; i++) c.QuadraticBezierTo(new Point(i * 70 + 35, 26), new Point(i * 70 + 70, 2), true, true);
            }
            g.Freeze();
            dc.DrawGeometry(null, wirePen, g);
            for (int i = 0; i < n; i++)
                for (int k = 1; k <= 3; k++)
                {
                    double t = k / 4.0, bx = i * 70 + t * 70, by = 2 * (1 - t) * (1 - t) + 2 * (1 - t) * t * 26 + t * t * 2;
                    double pulse = 0.5 + 0.5 * Math.Sin(time * 2.4 + i * 1.3 + k * 2.1);
                    int ci = (i * 3 + k) % BulbColors.Length;
                    dc.PushOpacity(0.25 + 0.75 * pulse);
                    dc.DrawEllipse(bulbHalo[ci], null, new Point(bx, by + 7), 14, 14);
                    dc.Pop();
                    dc.PushOpacity(0.6 + 0.4 * pulse);
                    dc.DrawEllipse(bulbCore[ci], null, new Point(bx, by + 7), 3.6, 5.2);
                    dc.Pop();
                }
        }

        static double Clamp(double v) { return v < 0 ? 0 : v > 1 ? 1 : v; }
        static double Ease(double t) { t = Clamp(t); return 1 - Math.Pow(1 - t, 3); }

        public void Toggle()
        {
            if (IsVisible) { Hide(); return; }
            if ((DateTime.Now - hiddenAt).TotalMilliseconds < 400) return;  // клик по значку сам закрыл панель
            foreach (Action a in refresh) a();
            Show();
            UpdateLayout();
            Rect wa = SystemParameters.WorkArea;
            Left = wa.Right - ActualWidth - 4;
            Top = wa.Bottom - ActualHeight - 4;
            Activate();
        }

        public static void Autostart(bool on)
        {
            using (Microsoft.Win32.RegistryKey k = Microsoft.Win32.Registry.CurrentUser.OpenSubKey(@"Software\Microsoft\Windows\CurrentVersion\Run", true))
            {
                if (k == null) return;
                if (on) k.SetValue("SnowDesk", "\"" + System.Reflection.Assembly.GetExecutingAssembly().Location + "\"");
                else k.DeleteValue("SnowDesk", false);
            }
        }

        public static bool AutostartOn()
        {
            using (Microsoft.Win32.RegistryKey k = Microsoft.Win32.Registry.CurrentUser.OpenSubKey(@"Software\Microsoft\Windows\CurrentVersion\Run"))
                return k != null && k.GetValue("SnowDesk") != null;
        }

        static WF.ToolStripMenuItem Item(string text, EventHandler click)
        {
            WF.ToolStripMenuItem i = new WF.ToolStripMenuItem(text);
            if (click != null) i.Click += click;
            return i;
        }

        static WF.ToolStripMenuItem Radio(string title, string[] names, Func<int> get, Action<int> set)
        {
            WF.ToolStripMenuItem root = new WF.ToolStripMenuItem(title);
            WF.ToolStripMenuItem[] items = new WF.ToolStripMenuItem[names.Length];
            for (int i = 0; i < names.Length; i++)
            {
                int v = i;
                items[i] = new WF.ToolStripMenuItem(names[i]);
                items[i].Checked = get() == i;
                items[i].Click += delegate { set(v); for (int k = 0; k < items.Length; k++) items[k].Checked = k == v; };
            }
            root.DropDownItems.AddRange(items);
            return root;
        }

        static WF.ToolStripMenuItem Check(string text, Func<bool> get, Action<bool> set)
        {
            WF.ToolStripMenuItem i = new WF.ToolStripMenuItem(text);
            i.Checked = get();
            i.Click += delegate { set(!get()); i.Checked = get(); };
            return i;
        }

        static System.Threading.Mutex mtx;
        static bool restart;

        public static void Log(Exception ex)
        {
            try { File.AppendAllText(Path.Combine(Settings.Dir, "error.txt"), DateTime.Now + "  " + ex + "\r\n\r\n"); } catch { }
        }

        public static void Quit()
        {
            tray.Visible = false;
            Application.Current.Shutdown();
        }

        public static void Restart()
        {
            restart = true;
            tray.Visible = false;
            Application.Current.Shutdown();
        }

        [STAThread]
        public static void Main()
        {
            AppDomain.CurrentDomain.UnhandledException += delegate (object o, UnhandledExceptionEventArgs e) { Log(e.ExceptionObject as Exception); };
            bool created;
            mtx = new System.Threading.Mutex(true, "Local\\SnowDeskOnce", out created);
            System.Threading.EventWaitHandle showPanel = new System.Threading.EventWaitHandle(false, System.Threading.EventResetMode.AutoReset, "Local\\SnowDeskPanel");
            if (!created) { showPanel.Set(); return; }   // снег уже идёт — ярлык просто открывает панель
            try { Native.SetProcessDPIAware(); } catch { }
            Application app = new Application();
            app.ShutdownMode = ShutdownMode.OnExplicitShutdown;
            app.DispatcherUnhandledException += delegate (object o, DispatcherUnhandledExceptionEventArgs e) { Log(e.Exception); e.Handled = true; };
            World w = new World();
            Settings S = w.S;
            w.Front = new Layer(w, true);
            w.Tip = delegate (string text) { tray.ShowBalloonTip(9000, "Снег на рабочем столе", text, WF.ToolTipIcon.Info); };

            WF.ContextMenuStrip menu = new WF.ContextMenuStrip();
            menu.Font = new SD.Font("Segoe UI", 10f);
            Panel panel = null;
            menu.Items.Add(Item("⚙  Открыть панель снега", delegate { if (panel != null) panel.Toggle(); }));
            menu.Items.Add(new WF.ToolStripSeparator());
            menu.Items.Add(Radio("❄  Снегопад", new string[] { "Лёгкий", "Ровный", "Метель" }, delegate { return S.Level; }, delegate (int v) { w.SetLevel(v); S.Save(); }));
            menu.Items.Add(Radio("🌬  Ветер", new string[] { "Порывами", "Штиль", "← Влево", "Вправо →" }, delegate { return S.Wind; }, delegate (int v) { S.Wind = v; S.Save(); }));
            menu.Items.Add(Radio("🖐  Курсор", new string[] { "Ничего не трогает", "🧤 Варежка — сметать снег", "⚪ Снежок — катать шары", "Лопата — сгребать много" }, delegate { return S.Tool; }, delegate (int v) { S.Tool = v; S.Save(); }));
            menu.Items.Add(new WF.ToolStripSeparator());
            menu.Items.Add(Item("⛄  Слепить снеговика", delegate { w.AutoSnowman(); }));
            menu.Items.Add(Item("🚜  Снегоуборщик — убрать снег", delegate { w.CleanAll(); }));
            menu.Items.Add(Item("🧹  Убрать весь снег мгновенно", delegate { w.Shovel(); }));
            menu.Items.Add(Item("🎅  Позвать Санту", delegate { w.CallSanta(); }));
            menu.Items.Add(Item("☃  Убрать снеговиков", delegate { w.ClearSnowmen(); }));
            menu.Items.Add(new WF.ToolStripSeparator());
            menu.Items.Add(Check("💡  Гирлянда", delegate { return S.Garland; }, delegate (bool v) { S.Garland = v; S.Save(); }));
            menu.Items.Add(Check("🪟  Сугробы на окнах", delegate { return S.Windows; }, delegate (bool v) { S.Windows = v; S.Save(); }));
            menu.Items.Add(Check("❄  Снег поверх окон", delegate { return S.Over; }, delegate (bool v) { S.Over = v; S.Save(); }));
            menu.Items.Add(Check("✨  Снег разлетается от мышки", delegate { return S.Mouse; }, delegate (bool v) { S.Mouse = v; S.Save(); }));
            menu.Items.Add(Check("🦌  Санта иногда пролетает", delegate { return S.Santa; }, delegate (bool v) { S.Santa = v; S.Save(); }));
            menu.Items.Add(new WF.ToolStripSeparator());
            menu.Items.Add(Check("⏸  Пауза", delegate { return S.Paused; }, delegate (bool v)
            {
                S.Paused = v;
                w.Front.Visibility = v ? Visibility.Hidden : Visibility.Visible;
            }));
            menu.Items.Add(Check("🚀  Запускать вместе с Windows", AutostartOn, Autostart));
            menu.Items.Add(Item("✖  Выход", delegate { tray.Visible = false; app.Shutdown(); }));

            tray = new WF.NotifyIcon();
            tray.Icon = MakeIcon();
            tray.Text = "Снег на рабочем столе";
            tray.ContextMenuStrip = menu;
            tray.Visible = true;
            panel = new Panel(w, delegate { tray.Visible = false; app.Shutdown(); });
            System.Threading.Thread waiter = new System.Threading.Thread(delegate ()
            {
                while (true) { showPanel.WaitOne(); app.Dispatcher.BeginInvoke(new Action(delegate { if (!panel.IsVisible) panel.Toggle(); })); }
            });
            waiter.IsBackground = true; waiter.Start();
            tray.MouseClick += delegate (object o, WF.MouseEventArgs e)
            {
                if (e.Button == WF.MouseButtons.Left) panel.Toggle();
            };

            w.Front.Show();
            // плавно: шаг на каждый кадр экрана
            CompositionTarget.Rendering += delegate { try { w.Step(); } catch (Exception ex) { Log(ex); } };
            if (S.Tips)
                tray.ShowBalloonTip(8000, "❄ Снег пошёл!", "Нажми на снежинку у часов — откроется панель: снегопад, варежка, снежки, лопата, снегоуборщик, Санта ⛄", WF.ToolTipIcon.Info);
            app.Run();
            tray.Visible = false;
            GC.KeepAlive(mtx);
            if (restart)
            {
                mtx.ReleaseMutex(); mtx.Dispose(); showPanel.Dispose();
                System.Diagnostics.Process.Start(System.Reflection.Assembly.GetExecutingAssembly().Location);
            }
        }
    }
