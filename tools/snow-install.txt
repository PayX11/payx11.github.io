    class Settings
    {
        public int Level = 1;          // 0 лёгкий, 1 обычный, 2 метель
        public bool Garland = true, Mouse = true, Windows = true, Santa = true, Paused = false;
        string path;
        public Settings()
        {
            string d = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "SnowDesk");
            Directory.CreateDirectory(d);
            path = Path.Combine(d, "settings.txt");
            try
            {
                foreach (string line in File.ReadAllLines(path))
                {
                    string[] kv = line.Split('=');
                    if (kv.Length != 2) continue;
                    switch (kv[0])
                    {
                        case "level": Level = Math.Max(0, Math.Min(2, int.Parse(kv[1]))); break;
                        case "garland": Garland = kv[1] == "1"; break;
                        case "mouse": Mouse = kv[1] == "1"; break;
                        case "windows": Windows = kv[1] == "1"; break;
                        case "santa": Santa = kv[1] == "1"; break;
                    }
                }
            }
            catch { }
        }
        public void Save()
        {
            try
            {
                File.WriteAllLines(path, new string[] {
                    "level=" + Level, "garland=" + (Garland ? 1 : 0), "mouse=" + (Mouse ? 1 : 0),
                    "windows=" + (Windows ? 1 : 0), "santa=" + (Santa ? 1 : 0) });
            }
            catch { }
        }
    }

        void MakeBrushes()
        {
            for (int i = 0; i < glow.Length; i++)
            {
                double a = 0.55 + i * 0.09;
                RadialGradientBrush g = new RadialGradientBrush();
                g.GradientStops.Add(new GradientStop(Color.FromArgb((byte)(255 * a), 255, 255, 255), 0));
                g.GradientStops.Add(new GradientStop(Color.FromArgb((byte)(200 * a), 240, 248, 255), 0.45));
                g.GradientStops.Add(new GradientStop(Color.FromArgb(0, 220, 235, 255), 1));
                g.Freeze();
                glow[i] = g;

        public void SetLevel(int l)
        {
            S.Level = l;
            int n = l == 0 ? 110 : l == 1 ? 240 : 520;
            while (flakes.Count < n) flakes.Add(NewFlake(true));
            if (flakes.Count > n) flakes.RemoveRange(n, flakes.Count - n);
        }

        Flake NewFlake(bool anywhere)
        {
            Flake f = new Flake();
            f.Depth = rnd.NextDouble();
            f.Crystal = rnd.NextDouble() < 0.16;
            f.R = (f.Crystal ? 4.5 : 1.2) + f.Depth * (f.Crystal ? 5.5 : 3.2);
            f.X = rnd.NextDouble() * (width + 200) - 100;
            f.Y = anywhere ? rnd.NextDouble() * height : -10 - rnd.NextDouble() * 60;
            f.Vy = 22 + f.Depth * 55 + rnd.NextDouble() * 18;
            f.Phase = rnd.NextDouble() * 6.28;
            f.Amp = 8 + rnd.NextDouble() * 22;
            f.Rot = rnd.NextDouble() * 360;
            f.RotSpeed = (rnd.NextDouble() - 0.5) * 90;
            return f;
        }

        double GroundAt(double x)
        {
            int i = (int)(x / COL);
            if (i < 0 || i >= ground.Length) return 0;
            return ground[i];
        }

        void AddSnow(double x, double amount)
        {
            int i = (int)(x / COL);
            for (int k = -2; k <= 2; k++)
            {
                int j = i + k;
                if (j < 0 || j >= ground.Length) continue;
                ground[j] = Math.Min(MaxGround(), ground[j] + amount * (k == 0 ? 0.5 : k == -1 || k == 1 ? 0.2 : 0.05));
            }
        }

        double MaxGround() { return 70; }

        void Step()
        {
            DateTime now = DateTime.Now;
            double dt = Math.Min(0.1, (now - last).TotalSeconds);
            last = now;
            if (S.Paused) { return; }
            time += dt;
            tick++;
            if (tick % 30 == 1) Housekeeping();
            if (hiddenFull) return;

            // ветер с порывами
            if (time > nextGust)
            {
                windTarget = (rnd.NextDouble() - 0.4) * (S.Level == 2 ? 140 : 60);
                nextGust = time + 3 + rnd.NextDouble() * 6;
            }
            wind += (windTarget - wind) * Math.Min(1, dt * 0.6);

            Native.POINT p;
            if (S.Mouse && Native.GetCursorPos(out p)) { mouseX = p.X / scale; mouseY = p.Y / scale; }
            else { mouseX = mouseY = -9999; }

            foreach (Flake f in flakes)
            {
                f.Phase += dt * (0.8 + f.Depth);
                double vx = wind * (0.4 + f.Depth * 0.8) + Math.Cos(f.Phase) * f.Amp * 0.6;
                // снег разлетается от мышки
                double dx = f.X - mouseX, dy = f.Y - mouseY, d2 = dx * dx + dy * dy;
                if (d2 < 120 * 120 && d2 > 1)
                {
                    double d = Math.Sqrt(d2), push = (1 - d / 120) * 420;
                    f.Vx += dx / d * push * dt;
                    f.Y += dy / d * push * dt * 0.5;
                }
                f.Vx *= Math.Pow(0.15, dt);
                f.X += (vx + f.Vx) * dt;
                f.Y += f.Vy * dt * (S.Level == 2 ? 1.35 : 1);
                f.Rot += f.RotSpeed * dt;
                if (f.X < -120) f.X += width + 220;
                if (f.X > width + 120) f.X -= width + 220;

                bool landed = false;
                if (f.Depth > 0.45)
                {
                    // шапки на окнах
                    if (S.Windows)
                    {
                        foreach (Cap c in caps.Values)
                        {
                            if (f.X < c.L || f.X > c.R) continue;
                            int ci = (int)((f.X - c.L) / COL);
                            if (ci < 0 || ci >= c.H.Length || c.H[ci] < 0) continue;
                            double top = c.T - c.H[ci];
                            if (f.Y >= top && f.Y < top + 14)
                            {
                                for (int k = -1; k <= 1; k++)
                                {
                                    int j = ci + k;
                                    if (j >= 0 && j < c.H.Length && c.H[j] >= 0)
                                        c.H[j] = Math.Min(16 + (j % 7 == 0 ? 3 : 0), c.H[j] + f.R * (k == 0 ? 1.0 : 0.4));
                                }
                                landed = true;
                                break;
                            }
                        }
                    }
                    if (!landed && f.Y >= groundY - GroundAt(f.X))
                    {
                        AddSnow(f.X, f.R * 2.0);
                        landed = true;
                    }
                }
                if (landed || f.Y > height + 20)
                {
                    Flake n = NewFlake(false);
                    f.X = n.X; f.Y = n.Y; f.Vy = n.Vy; f.Vx = 0; f.R = n.R; f.Depth = n.Depth; f.Crystal = n.Crystal;
                }
            }

            // сугроб оседает плавно, без иголок
            for (int i = 1; i < ground.Length - 1; i++)
            {
                double avg = (ground[i - 1] + ground[i + 1]) * 0.5;
                if (Math.Abs(ground[i] - avg) > 1.2) ground[i] += (avg - ground[i]) * 0.25;
            }
            foreach (Cap c in caps.Values)
            {
                for (int i = 1; i < c.H.Length - 1; i++)
                {
                    if (c.H[i] < 0 || c.H[i - 1] < 0 || c.H[i + 1] < 0) continue;
                    double avg = (c.H[i - 1] + c.H[i + 1]) * 0.5;
                    c.H[i] += (avg - c.H[i]) * 0.2;
                }
            }

            // комочки
            for (int i = chunks.Count - 1; i >= 0; i--)
            {
                Chunk c = chunks[i];
                c.Vy += 600 * dt;
                c.X += c.Vx * dt; c.Y += c.Vy * dt;
                c.Life -= dt;
                if (c.Y >= groundY - GroundAt(c.X)) { AddSnow(c.X, c.R * 0.3); chunks.RemoveAt(i); continue; }
                if (c.Life <= 0 || c.Y > height) chunks.RemoveAt(i);
            }

            // трактор
            if (!double.IsNaN(tractorX))
            {
                tractorX += 210 * dt;
                double plow = tractorX + 70;
                int i0 = (int)(plow / COL);
                for (int k = 0; k < 6; k++)
                {
                    int j = i0 + k;
                    if (j < 0 || j >= ground.Length) continue;
                    double take = ground[j] * 0.9;
                    if (take > 0.5 && rnd.NextDouble() < 0.35)
                        chunks.Add(new Chunk { X = j * COL, Y = groundY - ground[j], Vx = 120 + rnd.NextDouble() * 160, Vy = -220 - rnd.NextDouble() * 160, R = 2 + rnd.NextDouble() * 3, Life = 1.2 });
                    ground[j] -= take;
                }
                foreach (Snowman m in men) if (Math.Abs(m.X - plow) < 30 && m.Melt == 0) m.Melt = 0.001;
                if (tractorX > width + 160) tractorX = double.NaN;
            }

            // снеговики тают после трактора
            for (int i = men.Count - 1; i >= 0; i--)
            {
                Snowman m = men[i];
                if (m.Melt > 0)
                {
                    m.Melt += dt * 0.8;
                    if (m.Melt > 1) { for (int k = 0; k < 12; k++) chunks.Add(new Chunk { X = m.X, Y = groundY - 40 * m.Scale, Vx = (rnd.NextDouble() - 0.5) * 300, Vy = -200 - rnd.NextDouble() * 200, R = 3 + rnd.NextDouble() * 4, Life = 1.5 }); men.RemoveAt(i); }
                }
                else if (time > m.WaveAt + 6 + rnd.NextDouble() * 10) m.WaveAt = time;
            }

            // Санта
            if (S.Santa && double.IsNaN(santaX) && time > nextSanta) CallSanta();
            if (!double.IsNaN(santaX))
            {
                santaX -= 260 * dt;
                if (rnd.NextDouble() < 0.6)
                    chunks.Add(new Chunk { X = santaX + 160, Y = santaY + 26, Vx = 30, Vy = -10, R = 1.6, Life = 0.9 });
                if (santaX < -420) { santaX = double.NaN; nextSanta = time + 420 + rnd.NextDouble() * 600; }
            }

            view.InvalidateVisual();
        }

        public void CallSanta()
        {
            santaX = width + 60;
            santaY = 70 + rnd.NextDouble() * Math.Max(40, height * 0.25);
        }

        // раз в секунду: где окна, не играет ли что-то на весь экран, держимся поверх
        void Housekeeping()
        {
            if (me == IntPtr.Zero) return;
            IntPtr fg = Native.GetForegroundWindow();
            Native.RECT fr;
            bool full = false;
            if (fg != IntPtr.Zero && fg != me && fg != Native.GetShellWindow() && Native.GetWindowRect(fg, out fr))
            {
                string cls = ClassOf(fg);
                full = cls != "WorkerW" && cls != "Progman" && fr.L <= 0 && fr.T <= 0 &&
                       fr.R >= (int)(width * scale) && fr.B >= (int)(height * scale);
            }
            if (full != hiddenFull)
            {
                hiddenFull = full;
                Visibility = full ? Visibility.Hidden : Visibility.Visible;
            }
            if (!full) Native.SetWindowPos(me, new IntPtr(-1), 0, 0, 0, 0, 0x0001 | 0x0002 | 0x0010);
            if (S.Windows && !full) ScanWindows(); else if (caps.Count > 0) { DropAllCaps(); }
        }

        static string ClassOf(IntPtr h)
        {
            StringBuilder sb = new StringBuilder(64);
            Native.GetClassName(h, sb, 64);
            return sb.ToString();
        }

        void ScanWindows()
        {
            foreach (Cap c in caps.Values) c.Seen = false;
            List<Native.RECT> above = new List<Native.RECT>();
            Native.EnumWindows(delegate (IntPtr h, IntPtr p)
            {
                if (h == me || !Native.IsWindowVisible(h) || Native.IsIconic(h)) return true;
                int cloaked;
                if (Native.DwmGetWindowAttribute(h, 14, out cloaked, 4) == 0 && cloaked != 0) return true;
                string cls = ClassOf(h);
                if (cls == "Progman" || cls == "WorkerW" || cls == "Shell_TrayWnd" || cls == "Shell_SecondaryTrayWnd") return true;
                Native.RECT r;
                if (Native.DwmGetWindowAttribute(h, 9, out r, 16) != 0) Native.GetWindowRect(h, out r);
                if (r.R - r.L < 160 * scale || r.B - r.T < 80 * scale) { above.Add(r); return true; }
                if (Native.GetWindowTextLength(h) == 0) { above.Add(r); return true; }
                double L = r.L / scale, T = r.T / scale, R = r.R / scale;
                if (T > 30 && T < groundY - 40 && L < width && R > 0)
                {
                    Cap c;
                    if (!caps.TryGetValue(h, out c) || Math.Abs(c.L - L) > 1 || Math.Abs(c.T - T) > 1 || Math.Abs(c.R - R) > 1)
                    {
                        if (c != null) Crumble(c);
                        c = new Cap { L = L, T = T, R = R, H = new double[(int)((R - L) / COL) + 1] };
                        caps[h] = c;
                    }
                    c.Seen = true;
                    // где верх окна закрыт окнами, которые выше по порядку, снег не копится
                    for (int i = 0; i < c.H.Length; i++)
                    {
                        int px = (int)((c.L + i * COL) * scale), py = r.T - 2;
                        bool covered = false;
                        foreach (Native.RECT a in above) if (px >= a.L && px < a.R && py >= a.T && py < a.B) { covered = true; break; }
                        if (covered) { if (c.H[i] > 0) Spill(c.L + i * COL, c.T, c.H[i]); c.H[i] = -1; }
                        else if (c.H[i] < 0) c.H[i] = 0;
                    }
                }
                above.Add(r);
                return true;
            }, IntPtr.Zero);
            List<IntPtr> gone = new List<IntPtr>();
            foreach (KeyValuePair<IntPtr, Cap> kv in caps) if (!kv.Value.Seen) gone.Add(kv.Key);
            foreach (IntPtr h in gone) { Crumble(caps[h]); caps.Remove(h); }
        }

        void Spill(double x, double y, double h)
        {
            if (h < 2 || chunks.Count > 600) return;
            chunks.Add(new Chunk { X = x, Y = y - h, Vx = (rnd.NextDouble() - 0.5) * 60, Vy = -40, R = 1.5 + h * 0.18, Life = 3 });
        }

        void Crumble(Cap c)  // окно сдвинули — снег осыпается
        {
            for (int i = 0; i < c.H.Length; i += 2)
                if (c.H[i] > 1.5) Spill(c.L + i * COL, c.T, c.H[i]);
        }

        void DropAllCaps()
        {
            foreach (Cap c in caps.Values) Crumble(c);
            caps.Clear();
        }

        public void Tractor()
        {
            if (double.IsNaN(tractorX)) tractorX = -180;
            DropAllCaps();
        }

        // слепить снеговика: нужен снег в сугробе
        public string BuildSnowman()
        {
            if (men.Count >= 6) return "Снеговиков уже шесть — места больше нет ☃";
            double best = -1, bestX = 0;
            for (int t = 0; t < 40; t++)
            {
                double x = 80 + rnd.NextDouble() * (width - 160);
                bool near = false;
                foreach (Snowman m in men) if (Math.Abs(m.X - x) < 110) near = true;
                if (near) continue;
                double sum = 0;
                for (double k = x - 120; k <= x + 120; k += COL) sum += GroundAt(k);
                if (sum > best) { best = sum; bestX = x; }
            }
            if (best < 600) return "Снега пока мало ❄ Подожди немного или включи «Метель»";
            for (double k = bestX - 120; k <= bestX + 120; k += COL)
            {
                int j = (int)(k / COL);
                if (j >= 0 && j < ground.Length) ground[j] *= 0.55;
            }
            men.Add(new Snowman { X = bestX, Scale = 0.85 + rnd.NextDouble() * 0.35, Born = time, WaveAt = time + 2.5, Hat = rnd.Next(HatColors.Length + 1), Scarf = rnd.Next(ScarfColors.Length) });
            return null;
        }

        public void ClearSnowmen() { foreach (Snowman m in men) if (m.Melt == 0) m.Melt = 0.001; }

        // ---------- рисование ----------
        public void Draw(DrawingContext dc)
        {
            if (S.Garland) DrawGarland(dc);
            // дальние снежинки
            foreach (Flake f in flakes) if (f.Depth < 0.45) DrawFlake(dc, f);
            if (!double.IsNaN(santaX)) DrawSanta(dc, santaX, santaY + Math.Sin(time * 2) * 8);
            foreach (Cap c in caps.Values) DrawCap(dc, c);
            DrawGround(dc);
            foreach (Snowman m in men) DrawSnowman(dc, m);
            if (!double.IsNaN(tractorX)) DrawTractor(dc, tractorX);
            foreach (Chunk c in chunks) dc.DrawEllipse(glow[5], null, new Point(c.X, c.Y), c.R * 1.6, c.R * 1.6);
            foreach (Flake f in flakes) if (f.Depth >= 0.45) DrawFlake(dc, f);
        }

        void DrawFlake(DrawingContext dc, Flake f)
        {
            if (f.Crystal)
            {
                dc.PushTransform(new MatrixTransform(Rot(f.Rot, f.R, f.X, f.Y)));
                dc.PushOpacity(0.55 + f.Depth * 0.45);
                dc.DrawGeometry(null, crystalPen, crystalGeo);
                dc.Pop(); dc.Pop();
                dc.DrawEllipse(glow[2], null, new Point(f.X, f.Y), f.R * 0.45, f.R * 0.45);
            }
            else
            {
                int gi = Math.Min(glow.Length - 1, (int)(f.Depth * glow.Length));
                dc.DrawEllipse(glow[gi], null, new Point(f.X, f.Y), f.R * 1.7, f.R * 1.7);
            }
        }

        static Matrix Rot(double deg, double s, double x, double y)
        {
            Matrix m = Matrix.Identity;
            m.Scale(s, s);
            m.Rotate(deg);
            m.Translate(x, y);
            return m;
        }

        void DrawGround(DrawingContext dc)
        {
            StreamGeometry g = new StreamGeometry();
            using (StreamGeometryContext c = g.Open())
            {
                c.BeginFigure(new Point(0, groundY + 2), true, true);
                for (int i = 0; i < ground.Length; i++) c.LineTo(new Point(i * COL, groundY - ground[i] + 1), true, true);
                c.LineTo(new Point(width, groundY + 2), true, false);
            }
            g.Freeze();
            dc.DrawGeometry(snowFill, snowEdge, g);
            // искорки на сугробе
            for (int i = 3; i < ground.Length; i += 9)
            {
                if (ground[i] < 4) continue;
                double tw = Math.Sin(time * 3 + i * 1.7);
                if (tw > 0.85) dc.DrawEllipse(glow[5], null, new Point(i * COL, groundY - ground[i] + 4), 2.2, 2.2);
            }
        }

        void DrawCap(DrawingContext dc, Cap c)
        {
            int i = 0;
            while (i < c.H.Length)
            {
                while (i < c.H.Length && c.H[i] <= 0.3) i++;
                if (i >= c.H.Length) break;
                int s = i;
                StreamGeometry g = new StreamGeometry();
                using (StreamGeometryContext k = g.Open())
                {
                    k.BeginFigure(new Point(c.L + s * COL, c.T + 1), true, true);
                    while (i < c.H.Length && c.H[i] > 0.3) { k.LineTo(new Point(c.L + i * COL, c.T - c.H[i]), true, true); i++; }
                    k.LineTo(new Point(c.L + (i - 1) * COL, c.T + 1), true, false);
                }
                g.Freeze();
                dc.DrawGeometry(capFill, snowEdge, g);
            }
        }

        void DrawSnowman(DrawingContext dc, Snowman m)
        {
            double age = time - m.Born;
            double s = m.Scale * (1 - Math.Min(1, m.Melt) * 0.9);
            double baseY = groundY - Math.Max(0, GroundAt(m.X) - 6);
            double r1 = 30 * s, r2 = 22 * s, r3 = 15 * s;
            // шары накатываются по очереди
            double p1 = Clamp(age / 0.8), p2 = Clamp((age - 0.8) / 0.6), p3 = Clamp((age - 1.4) / 0.5), p4 = Clamp((age - 1.9) / 0.5);
            Brush body = glow[5];
            double y1 = baseY - r1, y2 = y1 - r1 - r2 * 0.75, y3 = y2 - r2 - r3 * 0.7;
            dc.PushOpacity(1 - Math.Min(1, m.Melt));
            if (p1 > 0) DrawBall(dc, m.X - (1 - Ease(p1)) * 140, y1, r1 * Ease(p1));
            if (p2 > 0) DrawBall(dc, m.X, y2 - (1 - Ease(p2)) * 60, r2 * Math.Max(0.3, Ease(p2)));
            if (p3 > 0) DrawBall(dc, m.X, y3 - (1 - Ease(p3)) * 80, r3 * Math.Max(0.3, Ease(p3)));
            if (p4 > 0)
            {
                dc.PushOpacity(Ease(p4));
                // руки-веточки, правая машет
                Pen arm = new Pen(stick, 2.4 * s); arm.StartLineCap = arm.EndLineCap = PenLineCap.Round;
                dc.DrawLine(arm, new Point(m.X - r2 * 0.8, y2), new Point(m.X - r2 * 2.0, y2 - r2 * 0.6));
                dc.DrawLine(arm, new Point(m.X - r2 * 1.6, y2 - r2 * 0.42), new Point(m.X - r2 * 1.75, y2 - r2 * 0.85));
                double wave = (time - m.WaveAt) < 2 ? Math.Sin((time - m.WaveAt) * 12) * 0.5 : 0;
                double ang = -0.55 + wave;
                Point sh = new Point(m.X + r2 * 0.8, y2);
                Point hand = new Point(sh.X + Math.Cos(ang) * r2 * 1.35, sh.Y + Math.Sin(ang) * r2 * 1.35);
                dc.DrawLine(arm, sh, hand);
                // глаза, улыбка, морковка
                dc.DrawEllipse(coal, null, new Point(m.X - r3 * 0.35, y3 - r3 * 0.2), 2.1 * s, 2.1 * s);
                dc.DrawEllipse(coal, null, new Point(m.X + r3 * 0.35, y3 - r3 * 0.2), 2.1 * s, 2.1 * s);
                for (int k = -2; k <= 2; k++)
                    dc.DrawEllipse(coal, null, new Point(m.X + k * r3 * 0.2, y3 + r3 * 0.38 + Math.Abs(k) * -r3 * 0.07), 1.2 * s, 1.2 * s);
                StreamGeometry nose = new StreamGeometry();
                using (StreamGeometryContext c = nose.Open())
                {
                    c.BeginFigure(new Point(m.X, y3 - r3 * 0.05), true, true);
                    c.LineTo(new Point(m.X + r3 * 1.2, y3 + r3 * 0.12), true, false);
                    c.LineTo(new Point(m.X, y3 + r3 * 0.2), true, false);
                }
                dc.DrawGeometry(carrot, null, nose);
                // пуговицы
                for (int k = 0; k < 3; k++) dc.DrawEllipse(coal, null, new Point(m.X, y2 - r2 * 0.35 + k * r2 * 0.4), 1.8 * s, 1.8 * s);
                // шарф с хвостом на ветру
                Brush sc = new SolidColorBrush(ScarfColors[m.Scarf]);
                dc.DrawRoundedRectangle(sc, null, new Rect(m.X - r3 * 1.05, y3 + r3 * 0.72, r3 * 2.1, 5 * s), 2.5 * s, 2.5 * s);
                double flap = Math.Sin(time * 5 + m.X) * 3 + wind * 0.03;
                StreamGeometry tail = new StreamGeometry();
                using (StreamGeometryContext c = tail.Open())
                {
                    c.BeginFigure(new Point(m.X + r3 * 0.3, y3 + r3 * 0.8), true, true);
                    c.LineTo(new Point(m.X + r3 * 0.7 + flap, y3 + r3 * 1.9), true, false);
                    c.LineTo(new Point(m.X + r3 * 1.1 + flap, y3 + r3 * 1.75), true, false);
                    c.LineTo(new Point(m.X + r3 * 0.75, y3 + r3 * 0.8), true, false);
                }
                dc.DrawGeometry(sc, null, tail);
                // шапка: цилиндр или ведро (последний вариант — без шапки)
                if (m.Hat < HatColors.Length)
                {
                    Brush hb = new SolidColorBrush(HatColors[m.Hat]);
                    if (m.Hat % 2 == 0)
                    {
                        dc.DrawRoundedRectangle(hb, null, new Rect(m.X - r3 * 1.05, y3 - r3 * 0.95, r3 * 2.1, 3.5 * s), 1.5, 1.5);
                        dc.DrawRectangle(hb, null, new Rect(m.X - r3 * 0.68, y3 - r3 * 2.0, r3 * 1.36, r3 * 1.1));
                        dc.DrawRectangle(new SolidColorBrush(ScarfColors[(m.Scarf + 2) % ScarfColors.Length]), null, new Rect(m.X - r3 * 0.68, y3 - r3 * 1.2, r3 * 1.36, 3 * s));
                    }
                    else
                    {
                        StreamGeometry b = new StreamGeometry();
                        using (StreamGeometryContext c = b.Open())
                        {
                            c.BeginFigure(new Point(m.X - r3 * 0.95, y3 - r3 * 0.7), true, true);
                            c.LineTo(new Point(m.X - r3 * 0.6, y3 - r3 * 1.75), true, false);
                            c.LineTo(new Point(m.X + r3 * 0.6, y3 - r3 * 1.75), true, false);
                            c.LineTo(new Point(m.X + r3 * 0.95, y3 - r3 * 0.7), true, false);
                        }
                        dc.DrawGeometry(hb, null, b);
                        dc.DrawEllipse(glow[5], null, new Point(m.X, y3 - r3 * 1.9), r3 * 0.35, r3 * 0.35);
                    }
                }
                dc.Pop();
            }
            dc.Pop();
        }

        void DrawBall(DrawingContext dc, double x, double y, double r)
        {

        }

        void DrawGarland(DrawingContext dc)
        {
            int n = (int)(width / 70);
            Pen wire = new Pen(new SolidColorBrush(Color.FromArgb(200, 30, 60, 30)), 1.6);
            StreamGeometry g = new StreamGeometry();
            using (StreamGeometryContext c = g.Open())
            {
                c.BeginFigure(new Point(0, 2), false, false);
                for (int i = 0; i < n; i++)
                {
                    double x0 = i * 70;
                    c.QuadraticBezierTo(new Point(x0 + 35, 26), new Point(x0 + 70, 2), true, true);
                }
            }
            dc.DrawGeometry(null, wire, g);
            for (int i = 0; i < n; i++)
            {
                for (int k = 1; k <= 3; k++)
                {
                    double t = k / 4.0, x0 = i * 70;
                    double bx = x0 + t * 70, by = 2 * (1 - t) * (1 - t) + 2 * (1 - t) * t * 26 + t * t * 2;
                    int ci = (i * 3 + k) % BulbColors.Length;
                    double pulse = 0.5 + 0.5 * Math.Sin(time * 2.4 + i * 1.3 + k * 2.1);
                    Color col = BulbColors[ci];
                    RadialGradientBrush halo = new RadialGradientBrush(Color.FromArgb((byte)(150 * pulse), col.R, col.G, col.B), Color.FromArgb(0, col.R, col.G, col.B));
                    dc.DrawEllipse(halo, null, new Point(bx, by + 7), 14, 14);
                    dc.DrawEllipse(new SolidColorBrush(Color.FromArgb((byte)(140 + 115 * pulse), col.R, col.G, col.B)), null, new Point(bx, by + 7), 3.6, 5.2);
                }
            }
        }

        static double Clamp(double v) { return v < 0 ? 0 : v > 1 ? 1 : v; }
        static double Ease(double t) { t = Clamp(t); return 1 - Math.Pow(1 - t, 3); }
    }
